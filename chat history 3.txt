The ComplaintController doesn't have a GetById action. That's why it 404s — the route doesn't exist. If you want a "view single complaint" endpoint (the frontend might need it for the detail view), add this to ComplaintController.cs:

csharp
/// <summary>Get one complaint (owner or admin).</summary>
[HttpGet("{id:guid}")]
public async Task<IActionResult> GetById(Guid id)
{
    var result = await _service.GetByIdAsync(id, CurrentUserId, IsAdmin);
    return result is null ? NotFound() : Ok(result);
}
And add to IComplaintService / ComplaintService:

csharp
// IComplaintService
Task<ComplaintResponse?> GetByIdAsync(Guid id, Guid requesterId, bool isAdmin);

// ComplaintService
public async Task<ComplaintResponse?> GetByIdAsync(Guid id, Guid requesterId, bool isAdmin)
{
    var c = await Query().FirstOrDefaultAsync(x => x.Id == id);
    if (c is null) return null;
    if (!isAdmin && c.TenantId != requesterId) return null;
    return Map(c);
}
If you skip it, the frontend can use /api/complaints/mine for tenants and /api/complaints for admins, so this is optional — just noting it since your test included it.

[ApiController]
[Route("api/maps")]
public class GoogleMapsController : ControllerBase
{
    private readonly IGoogleMapsService _service;
    public GoogleMapsController(IGoogleMapsService service) => _service = service;

    // Geocode an address to lat/lng.
    [HttpGet("geocode")]
    public async Task<IActionResult> Geocode([FromQuery] string address)
    {
        if (string.IsNullOrWhiteSpace(address))
            return BadRequest(new { error = "Address is required." });

        var result = await _service.GeocodeAsync(address);
        if (result is null)
            return StatusCode(StatusCodes.Status503ServiceUnavailable, new { error = "Geocoding service is unavailable." });
        return Ok(result);
    }

    // Nearby places (universities, shops, etc.).
    [HttpGet("nearby")]
    public async Task<IActionResult> Nearby([FromQuery] double lat, [FromQuery] double lng,
                                             [FromQuery] string type = "university",
                                             [FromQuery] int radius = 3000)
    {
        var results = await _service.GetNearbyAsync(lat, lng, type, radius);
        return Ok(results);
    }
}

public static class ClaimsPrincipalExtensions
{
    /// <summary>
    /// Safely reads the user id from the JWT claims. Returns Guid.Empty when the claim
    /// is missing or malformed - never throws.
    /// </summary>
    public static Guid GetUserId(this ClaimsPrincipal principal)
    {
        var claim = principal.FindFirstValue(ClaimTypes.NameIdentifier)
                    ?? principal.FindFirstValue("sub");

        return Guid.TryParse(claim, out var id) ? id : Guid.Empty;
    }

/// Assigns or propagates an X-Correlation-Id on every request.
/// The ID is added to the log scope so every log line for a request is traceable.
public class CorrelationIdMiddleware
{
    private const string HeaderName = "X-Correlation-Id";
    private readonly RequestDelegate _next;
    private readonly ILogger<CorrelationIdMiddleware> _logger;

    public CorrelationIdMiddleware(RequestDelegate next, ILogger<CorrelationIdMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        var correlationId = context.Request.Headers[HeaderName].FirstOrDefault();

        if (string.IsNullOrWhiteSpace(correlationId))
            correlationId = Guid.NewGuid().ToString("N");

        context.Items[HeaderName] = correlationId;
        context.Response.Headers[HeaderName] = correlationId;
        context.TraceIdentifier = correlationId;

        using (_logger.BeginScope(new Dictionary<string, object>
        {
            ["CorrelationId"] = correlationId
        }))
        {
            await _next(context);
        }
    }
}

        _logger = logger;
    }

    public async Task InvokeAsync(HttpContext context)

                context.Request.Path,
                context.Response.StatusCode,
                sw.ElapsedMilliseconds);
        }
    }
}

// Adds hardened security response headers.
public class SecurityHeadersMiddleware
{
    private readonly RequestDelegate _next;
    private readonly IWebHostEnvironment _env;

    public SecurityHeadersMiddleware(RequestDelegate next, IWebHostEnvironment env)
    {
        _next = next;
        _env = env;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        var headers = context.Response.Headers;

        headers["X-Content-Type-Options"] = "nosniff";
        headers["X-Frame-Options"] = "DENY";
        headers["X-XSS-Protection"] = "1; mode=block";
        headers["Referrer-Policy"] = "strict-origin-when-cross-origin";
        headers["Permissions-Policy"] = "geolocation=(), microphone=(), camera=()";

        if (!_env.IsDevelopment())
        {
            headers["Content-Security-Policy"] =
                "default-src 'self'; " +
                "img-src 'self' data: https:; " +
                "script-src 'self' https://maps.googleapis.com; " +
                "style-src 'self' 'unsafe-inline'; " +
                "font-src 'self' data:; " +
                "connect-src 'self' https://maps.googleapis.com; " +
                "frame-ancestors 'none'";
        }

        headers.Remove("X-Powered-By");
        headers.Remove("Server");

        await _next(context);
    }
}

/// <summary>
/// Sends notifications by email using SendGrid.
/// </summary>
public class EmailNotificationStrategy : INotificationStrategy
{
    private readonly IConfiguration _config;
    private readonly ILogger<EmailNotificationStrategy> _logger;

    public EmailNotificationStrategy(
        IConfiguration config,
        ILogger<EmailNotificationStrategy> logger)
    {
        _config = config;
        _logger = logger;
    }

    /// <inheritdoc />
    public async Task SendAsync(string recipient, string subject, string message)
    {
        var apiKey = _config["SendGrid:ApiKey"];
        var fromEmail = _config["SendGrid:FromEmail"] ?? "noreply@unicribz.co.za";
        var fromName = _config["SendGrid:FromName"] ?? "UniCribz";

        if (string.IsNullOrWhiteSpace(apiKey))
        {
            _logger.LogWarning(
                "SendGrid API key is not configured — skipping email to {Recipient}",
                recipient);
            return;
        }

        var client = new SendGridClient(apiKey);
        var from = new EmailAddress(fromEmail, fromName);
        var to = new EmailAddress(recipient);

        var msg = MailHelper.CreateSingleEmail(
            from,
            to,
            subject,
            plainTextContent: message,
            htmlContent: $"<p>{System.Net.WebUtility.HtmlEncode(message)}</p>");

        var response = await client.SendEmailAsync(msg);

        if (!response.IsSuccessStatusCode)
        {
            var body = await response.Body.ReadAsStringAsync();
            throw new InvalidOperationException(
                $"SendGrid failed with status {response.StatusCode}: {body}");
        }
    }
}

/// <summary>
/// Orchestrates delivery of a notification across all registered
/// <see cref="INotificationStrategy"/> implementations.
/// Callers interact with this abstraction only — they do not need to know
/// which channels (email, SMS, in-app) are active.
/// </summary>
public interface INotificationManager
{
    /// <summary>
    /// Sends the notification through every registered <see cref="INotificationStrategy"/>.
    /// Failures in one channel are logged and swallowed so that other channels still run.
    /// </summary>
    Task NotifyAsync(string recipient, string subject, string message);
}
namespace UniCribz.Api.Notifications;

/// <summary>
/// Defines the contract for a single notification channel.
/// Implements the Strategy Pattern — each concrete strategy handles one delivery mechanism
/// (email, SMS, in-app, etc.) and can be swapped or combined without changing the caller.
/// </summary>
public interface INotificationStrategy
{
    /// <summary>
    /// Sends a notification to the specified recipient via this strategy's channel.
    /// </summary>
    /// <param name="recipient">
    /// The destination address for the notification. For email this is an email address,
    /// for SMS this is a phone number in E.164 format, and for in-app this is the user's ID.
    /// </param>
    /// <param name="subject">A short subject or title for the notification.</param>
    /// <param name="message">The full body of the notification.</param>
    Task SendAsync(string recipient, string subject, string message);
}
using Microsoft.Extensions.Logging;

namespace UniCribz.Api.Notifications;

/// <summary>
/// Default implementation of <see cref="INotificationManager"/>.
/// Iterates over every registered <see cref="INotificationStrategy"/> and dispatches the notification.
/// </summary>
public class NotificationManager : INotificationManager
{
    private readonly IEnumerable<INotificationStrategy> _strategies;
    private readonly ILogger<NotificationManager> _logger;

    public NotificationManager(
        IEnumerable<INotificationStrategy> strategies,
        ILogger<NotificationManager> logger)
    {
        _strategies = strategies;
        _logger = logger;
    }

    /// <inheritdoc />
    public async Task NotifyAsync(string recipient, string subject, string message)
    {
        foreach (var strategy in _strategies)
        {
            try
            {
                await strategy.SendAsync(recipient, subject, message);
                _logger.LogInformation(
                    "Notification sent via {Strategy} to {Recipient}",
                    strategy.GetType().Name,
                    recipient);
            }
            catch (Exception ex)
            {
                // Do not let one channel failure break the others.
                _logger.LogError(
                    ex,
                    "Notification failed via {Strategy} to {Recipient}",
                    strategy.GetType().Name,
                    recipient);
            }
        }
    }
}

/// <summary>
/// Sends notifications by SMS using Twilio.
/// </summary>
public class SMSNotificationStrategy : INotificationStrategy
{
    private readonly IConfiguration _config;
    private readonly ILogger<SMSNotificationStrategy> _logger;

    public SMSNotificationStrategy(
        IConfiguration config,
        ILogger<SMSNotificationStrategy> logger)
    {
        _config = config;
        _logger = logger;
    }

    /// <inheritdoc />
    public async Task SendAsync(string recipient, string subject, string message)
    {
        var accountSid = _config["Twilio:AccountSid"];
        var authToken = _config["Twilio:AuthToken"];
        var fromNumber = _config["Twilio:FromNumber"];

        if (string.IsNullOrWhiteSpace(accountSid) ||
            string.IsNullOrWhiteSpace(authToken) ||
            string.IsNullOrWhiteSpace(fromNumber))
        {
            _logger.LogWarning(
                "Twilio is not configured — skipping SMS to {Recipient}",
                recipient);
            return;
        }

        TwilioClient.Init(accountSid, authToken);

        var body = string.IsNullOrWhiteSpace(subject)
            ? message
            : $"{subject}: {message}";

        if (body.Length > 1600)
        {
            body = body.Substring(0, 1600);
        }

        await MessageResource.CreateAsync(
            to: new PhoneNumber(recipient),
            from: new PhoneNumber(fromNumber),
            body: body);
    }
}

// Observer that frees the room when a tenant moves out.
// On MOVE_OUT: room status → Available, timestamped.
public class RoomStatusUpdater : IStatusObserver
{
    private readonly UniCribzDbContext _db;
    private readonly ILogger<RoomStatusUpdater> _logger;

    public RoomStatusUpdater(UniCribzDbContext db, ILogger<RoomStatusUpdater> logger)
    {
        _db = db;
        _logger = logger;
    }

    public async Task UpdateAsync(StatusChangeEvent change)
    {
        if (change.EventType != "MOVE_OUT")
            return;

        var room = await _db.Rooms.FindAsync(change.RoomId);
        if (room is null)
        {
            _logger.LogWarning(
                "[RoomStatusUpdater] Room {RoomId} not found for MOVE_OUT of tenant {TenantId}",
                change.RoomId, change.TenantId);
            return;
        }

        room.Status = RoomStatus.Available;
        room.UpdatedAt = DateTime.UtcNow;
        await _db.SaveChangesAsync();

        _logger.LogInformation(
            "[RoomStatusUpdater] Room {RoomNumber} marked Available after tenant {TenantId} moved out",
            room.RoomNumber, change.TenantId);
    }
}

    {
        var list = await _db.Notifications
            .Where(n => n.UserId == userId)
            .OrderByDescending(n => n.CreatedAt)
            .ToListAsync();

        return list.Select(Map);
    }

    public async Task<NotificationResponse> MarkAsReadAsync(Guid id, Guid userId)
    {
        var n = await _db.Notifications.FindAsync(id)
            ?? throw new KeyNotFoundException("Notification not found.");

        if (n.UserId != userId)
            throw new UnauthorizedAccessException("You can only read your own notifications.");

        n.IsRead = true;
        n.ReadAt = DateTime.UtcNow;
        await _db.SaveChangesAsync();
        return Map(n);
    }

    public Task<int> GetUnreadCountAsync(Guid userId) =>
        _db.Notifications.CountAsync(n => n.UserId == userId && !n.IsRead);

    private static NotificationResponse Map(Data.Entities.Notification n) => new()
    {
        Id = n.Id,
        Title = n.Title,
        Message = n.Message,
        Channel = n.Channel,
        Severity = n.Severity,
        IsRead = n.IsRead,
        ReadAt = n.ReadAt,
        RelatedEntityType = n.RelatedEntityType,
        RelatedEntityId = n.RelatedEntityId,
        CreatedAt = n.CreatedAt
    };

public class PropertyRepository : Repository<Property>, IPropertyRepository
{
    public PropertyRepository(UniCribzDbContext db) : base(db) { }

    public async Task<Property?> GetWithDetailsAsync(Guid id)
        => await Set
            .Include(p => p.Rooms)
            .Include(p => p.Images)
            .Include(p => p.Amenities)
            .FirstOrDefaultAsync(p => p.Id == id);

    public async Task<IEnumerable<Property>> SearchAsync(
        string? city = null,
        string? suburb = null,
        string? propertyType = null,
        decimal? maxRent = null,
        Guid? nearUniversityId = null)
    {
        var query = Set
            .AsNoTracking()
            .Include(p => p.Rooms)
            .Include(p => p.Amenities)
            .Where(p => p.IsActive);

        if (!string.IsNullOrWhiteSpace(city))
            query = query.Where(p => p.City.ToLower() == city.ToLower());

        if (!string.IsNullOrWhiteSpace(suburb))
            query = query.Where(p => p.Suburb.ToLower() == suburb.ToLower());

        if (!string.IsNullOrWhiteSpace(propertyType))
            query = query.Where(p => p.PropertyType == propertyType);

        if (maxRent.HasValue)
            query = query.Where(p => p.Rooms.Any(r => r.MonthlyRent <= maxRent.Value));

        
        if (nearUniversityId.HasValue)
        {
            var uni = await Db.Universities.AsNoTracking()
                                           .FirstOrDefaultAsync(u => u.Id == nearUniversityId.Value);
            if (uni is not null)
                query = query.Where(p => p.City == uni.City);
        }

        return await query.ToListAsync();
    }
}

/// <summary>
/// EF Core implementation of <see cref="IRepository{T}"/>.
/// Never calls SaveChanges automatically — services decide when to commit.
/// </summary>
public class Repository<T> : IRepository<T> where T : class
{
    protected readonly UniCribzDbContext Db;
    protected readonly DbSet<T> Set;

    public Repository(UniCribzDbContext db)
    {
        Db = db;
        Set = db.Set<T>();
    }

    public virtual async Task<T?> GetByIdAsync(Guid id)
        => await Set.FindAsync(id);

    public virtual async Task<IEnumerable<T>> GetAllAsync()
        => await Set.AsNoTracking().ToListAsync();

    public virtual async Task<IEnumerable<T>> FindAsync(Expression<Func<T, bool>> predicate)
        => await Set.AsNoTracking().Where(predicate).ToListAsync();

    public virtual async Task<T> AddAsync(T entity)
    {
        await Set.AddAsync(entity);
        return entity;
    }

    public virtual async Task AddRangeAsync(IEnumerable<T> entities)
        => await Set.AddRangeAsync(entities);

    public virtual void Update(T entity)
        => Set.Update(entity);

    public virtual void Remove(T entity)
        => Set.Remove(entity);

    public virtual async Task DeleteAsync(Guid id)
    {
        var entity = await Set.FindAsync(id);
        if (entity is not null)
            Set.Remove(entity);
    }

    public virtual async Task<int> CountAsync()
        => await Set.CountAsync();

    public virtual async Task<int> CountAsync(Expression<Func<T, bool>> predicate)
        => await Set.CountAsync(predicate);

    public virtual async Task<bool> ExistsAsync(Guid id)
        => await Set.FindAsync(id) is not null;

    public virtual async Task<int> SaveChangesAsync()
        => await Db.SaveChangesAsync();
}

namespace UniCribz.Shared.Helpers;

public static class JwtTokenGenerator
{
    public static (string Token, DateTime ExpiresAt) Generate(
        Guid userId,
        string email,
        string fullName,
        string role,
        IConfiguration config)
    {
        var key = config["Jwt:Key"]
                  ?? throw new InvalidOperationException("Jwt:Key is not configured.");
        var issuer = config["Jwt:Issuer"] ?? "UniCribz";
        var audience = config["Jwt:Audience"] ?? "UniCribzClients";
        var expiryMinutes = int.TryParse(config["Jwt:ExpiryMinutes"], out var m) ? m : 60;

        var expiresAt = DateTime.UtcNow.AddMinutes(expiryMinutes);

        var claims = new[]
        {
            new Claim(JwtRegisteredClaimNames.Sub, userId.ToString()),
            new Claim(ClaimTypes.NameIdentifier, userId.ToString()),
            new Claim(JwtRegisteredClaimNames.Email, email),
            new Claim(ClaimTypes.Email, email),
            new Claim(JwtRegisteredClaimNames.Name, fullName),
            new Claim(ClaimTypes.Role, role),
            new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString())
        };

        var signingKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(key));
        var credentials = new SigningCredentials(signingKey, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: issuer,
            audience: audience,
            claims: claims,
            expires: expiresAt,
            signingCredentials: credentials);

        return (new JwtSecurityTokenHandler().WriteToken(token), expiresAt);
    }
}

public static class PasswordHasher
{
    // Hashes a password using BCrypt with a work factor of 12.
    public static string Hash(string password)
    {
        if (string.IsNullOrWhiteSpace(password))
            throw new ArgumentException("Password cannot be empty.", nameof(password));

        return BCrypt.Net.BCrypt.HashPassword(password, workFactor: 12);
    }

    // Verifies a password against a given hash.
    public static bool Verify(string password, string hash)
    {
        if (string.IsNullOrWhiteSpace(password) || string.IsNullOrWhiteSpace(hash))
            return false;

        return BCrypt.Net.BCrypt.Verify(password, hash);
    }
}

        // each card costs one extra API call (for the price), so cap the page
        var cards = await PropertyCards.BuildAsync(_api, properties.OrderBy(p => p.Name).Take(12));

        });

    {
        var property = await _api.GetPropertyAsync(id);
        if (property is null)
            return NotFound();

        var rooms = await _api.GetRoomsAsync(id) ?? new();

        return View(new PropertyDetailsViewModel { Property = property, Rooms = rooms });
    }

/// <summary>
/// Catches ApiException thrown by page loads. An expired/invalid token sends the user to the
/// login page; anything else shows a friendly error page. (Form posts handle their own
/// ApiExceptions so the form can be redisplayed with the message.)
/// </summary>
public class ApiExceptionFilter : IAsyncExceptionFilter
{
    public async Task OnExceptionAsync(ExceptionContext context)
    {
        if (context.Exception is not ApiException ex)
            return;

        context.ExceptionHandled = true;

        if (ex.StatusCode == 401)
        {
            await context.HttpContext.SignOutAsync();
            var request = context.HttpContext.Request;
            context.Result = new RedirectToActionResult("Login", "Account", new
            {
                returnUrl = request.Method == "GET" ? request.Path + request.QueryString : null,
                expired = true
            });
            return;
        }

        var viewData = new ViewDataDictionary<ApiException>(new EmptyModelMetadataProvider(), context.ModelState)
        {
            Model = ex
        };

        context.Result = new ViewResult
        {
            ViewName = "ApiError",
            ViewData = viewData,
            StatusCode = ex.StatusCode == 404 ? 404 : 502
        };
    }
}

/// <summary>
/// The API's property response has no price, so each card looks up its rooms and shows
/// the lowest rent (preferring rooms that are currently available).
/// </summary>
public static class PropertyCards
{
    public static async Task<List<PropertyCardViewModel>> BuildAsync(IApiClient api, IEnumerable<PropertyDto> properties)
    {
        var tasks = properties.Select(async property =>
        {
            decimal? minRent = null;
            try
            {
                var rooms = await api.GetRoomsAsync(property.Id) ?? new List<RoomDto>();
                var pool = rooms.Where(r => r.Status == "Available").ToList();
                if (pool.Count == 0) pool = rooms;
                if (pool.Count > 0) minRent = pool.Min(r => r.MonthlyRent);
            }
            catch (ApiException)
            {
                // a failed price lookup shouldn't hide the property
            }

            return new PropertyCardViewModel { Property = property, MinRent = minRent };
        });

        return (await Task.WhenAll(tasks)).ToList();
    }
}
